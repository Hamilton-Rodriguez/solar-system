
    slack-notification:
      if: always()
      name: Slack Notification
      needs: [dev-integration-test, prod-integration-test]
      continue-on-error: true
      runs-on: ubuntu-latest
      steps:
        - uses: actions/checkout@v4
        - name: Slack Notification
          uses: ftCamp/action-slack-notify@v2
          env:
            SLACK_WEBHOOK: ${{ secrets.SLACK_WEBHOOK }}
            SLACK_CHANNEL: ${{ vars.SLACK_CHANNEL }}
            SLACK_COLOR: ${{ job.status}} # or a specific color code like 'good' or '#ff000...'
            SLACK_ICON: https://gitlab.com/sidd-harth/solar-system/-/raw/main/images/saturn.png
            SLACK_MESSAGE: ':hammer_and_wrench: Triggered by ${{ github.actor }}'
